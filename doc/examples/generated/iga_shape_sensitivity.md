# iga_shape_sensitivity Example

# Isogeometric shape sensitivity and adjoint optimization

This example differentiates a complete scalar isogeometric state problem,

\[
K(P,w)u=f,\qquad
\mathcal J(P,w)=\tfrac12\lVert u-u_\star\rVert_2^2,
\]

with respect to every NURBS control point \(P\) and rational weight \(w\).
Forward mode propagates one prescribed mesh-motion direction. Reverse mode
uses one sparse adjoint solve and returns the full shape gradient, making its
cost independent of the number of geometry design variables.

The local rational-geometry and H1 element products are generated by FortSym.
The sparse solve uses exact implicit tangent and adjoint equations, so the
production path works with GCC and does not differentiate a third-party
factorization. A complete reassemble-and-refactor central difference is used
only as an independent diagnostic and timing baseline.

Generated gallery artifacts:

- `shape_gradient_2d.png`: original control mesh and an exaggerated descent step;
- `state_sensitivity_1d.png`: state and directional state derivative;
- `geometry_gradient_1d.png`: gradient magnitude for every geometry DOF;
- `benchmark.txt`: analytical JVP/VJP and finite-difference timings and errors.

No rendered image is committed.

## Provenance

- J. Nocedal and S. Wright, *Numerical Optimization*, 2nd ed., Springer
  (2006), adjoint/reduced-gradient methodology.
- J. Cottrell, T. Hughes, and Y. Bazilevs, *Isogeometric Analysis*, Wiley
  (2009), rational spline geometry and Galerkin discretization.
- A. Buffa, J. Rivas, G. Sangalli, and R. Vázquez, *Isogeometric discrete
  differential forms in three dimensions*, SIAM J. Numer. Anal. 49 (2011),
  <https://doi.org/10.1137/100786708>.

## Usage

```bash
fpm run --example iga_shape_sensitivity
```

## Source Code

```fortran
program iga_shape_sensitivity
    use fortfem_api, only: assemble_bspline_h1_operator_csc, &
        assemble_bspline_h1_operator_csc_jvp, &
        assemble_bspline_h1_operator_csc_vjp, sparse_direct_factor_csc, &
        sparse_direct_factor_t, sparse_direct_factor_transpose_csc, &
        sparse_direct_free, sparse_direct_solve_factored, &
        sparse_direct_solve_factored_jvp, sparse_direct_solve_factored_vjp
    use fortfem_kinds, only: dp
    use fortplot, only: figure, legend, plot, savefig, set_yscale, title, &
        xlabel, ylabel
    use fortsparse, only: csc_t, fortsparse_status_t
    implicit none

    integer, parameter :: degree = 2, quadrature_order = 4
    integer, parameter :: repetitions = 20
    real(dp), parameter :: knots(7) = [ &
        0.0_dp, 0.0_dp, 0.0_dp, 0.5_dp, 1.0_dp, 1.0_dp, 1.0_dp]
    real(dp), parameter :: difference_step = 1.0e-6_dp
    real(dp), parameter :: blue(3) = [0.0_dp, 114.0_dp, 178.0_dp]/255.0_dp
    real(dp), parameter :: orange(3) = [230.0_dp, 159.0_dp, 0.0_dp]/255.0_dp
    character(*), parameter :: output_directory = &
        "output/example/iga_shape_sensitivity"
    real(dp), allocatable :: control_points(:, :, :)
    real(dp), allocatable :: control_points_bar(:, :, :)
    real(dp), allocatable :: control_points_dot(:, :, :)
    real(dp), allocatable :: control_points_minus(:, :, :)
    real(dp), allocatable :: control_points_plus(:, :, :)
    real(dp), allocatable :: displayed_points(:, :, :)
    real(dp), allocatable :: geometry_gradient(:), weights(:, :)
    real(dp), allocatable :: weights_bar(:, :), weights_dot(:, :)
    real(dp), allocatable :: weights_minus(:, :), weights_plus(:, :)
    real(dp), allocatable :: rhs(:), rhs_bar(:), rhs_dot(:)
    real(dp), allocatable :: solution(:), solution_bar(:), solution_dot(:)
    real(dp), allocatable :: solution_minus(:), solution_plus(:), target(:)
    real(dp), allocatable :: matrix_values_bar(:), indices(:)
    real(dp), allocatable :: x_points(:), y_points(:)
    real(dp) :: adjoint_derivative, analytical_jvp_seconds
    real(dp) :: analytical_vjp_seconds, central_derivative
    real(dp) :: finite_difference_seconds, maximum_gradient, objective
    real(dp) :: start_time
    type(csc_t) :: matrix, matrix_bar, matrix_dot, matrix_minus, matrix_plus
    type(fortsparse_status_t) :: sparse_status
    type(sparse_direct_factor_t) :: factor, factor_minus, factor_plus
    type(sparse_direct_factor_t) :: transpose_factor
    integer :: i, ix, iy, status, unit

    call execute_command_line("mkdir -p "//output_directory)
    x_points = greville_abscissae(knots, degree)
    y_points = greville_abscissae(knots, degree)
    allocate (control_points(2, size(x_points), size(y_points)))
    allocate (weights(size(x_points), size(y_points)))
    do iy = 1, size(y_points)
        do ix = 1, size(x_points)
            control_points(:, ix, iy) = [ &
                1.0_dp + 1.2_dp*x_points(ix), &
                0.8_dp*y_points(iy) + 0.1_dp*x_points(ix)]
            weights(ix, iy) = 1.0_dp + &
                0.05_dp*sin(real(3*ix + 5*iy, dp))
        end do
    end do
    allocate (control_points_dot, mold=control_points)
    allocate (weights_dot, mold=weights)
    control_points_dot = reshape([ &
        (0.1_dp*sin(real(7*i + 2, dp)), i=1, size(control_points_dot))], &
        shape(control_points_dot))
    weights_dot = reshape([ &
        (0.03_dp*cos(real(5*i - 1, dp)), i=1, size(weights_dot))], &
        shape(weights_dot))

    call assemble_primal(control_points, weights, matrix)
    allocate (rhs(matrix%nrow), rhs_dot(matrix%nrow))
    allocate (solution(matrix%nrow), solution_dot(matrix%nrow))
    allocate (solution_bar(matrix%nrow), rhs_bar(matrix%nrow))
    allocate (target(matrix%nrow), matrix_values_bar(matrix%nnz))
    rhs = [(1.0_dp + 0.1_dp*cos(real(2*i, dp)), i=1, matrix%nrow)]
    rhs_dot = 0.0_dp
    target = [(0.2_dp*sin(real(i, dp)), i=1, matrix%nrow)]
    call sparse_direct_factor_csc( &
        factor, matrix%nrow, matrix%col_ptr, matrix%row_idx, matrix%val, status)
    if (status /= 0) error stop "primal factorization failed"
    call sparse_direct_solve_factored(factor, rhs, solution, status)
    if (status /= 0) error stop "primal solve failed"
    solution_bar = solution - target
    objective = 0.5_dp*dot_product(solution_bar, solution_bar)

    call cpu_time(start_time)
    do i = 1, repetitions
        call assemble_tangent(matrix_dot)
    end do
    call cpu_time(analytical_jvp_seconds)
    analytical_jvp_seconds = &
        (analytical_jvp_seconds - start_time)/real(repetitions, dp)
    call sparse_direct_solve_factored_jvp( &
        factor, matrix%nrow, matrix_dot%col_ptr, matrix_dot%row_idx, &
        matrix_dot%val, solution, rhs_dot, solution_dot, status)
    if (status /= 0) error stop "state tangent solve failed"

    call sparse_direct_factor_transpose_csc( &
        transpose_factor, matrix%nrow, matrix%col_ptr, matrix%row_idx, &
        matrix%val, status)
    if (status /= 0) error stop "adjoint factorization failed"
    call sparse_direct_solve_factored_vjp( &
        transpose_factor, matrix%nrow, matrix%col_ptr, matrix%row_idx, &
        solution, solution_bar, rhs_bar, matrix_values_bar, status)
    if (status /= 0) error stop "state adjoint solve failed"
    matrix_bar = matrix
    matrix_bar%val = matrix_values_bar
    allocate (control_points_bar, mold=control_points)
    allocate (weights_bar, mold=weights)
    call cpu_time(start_time)
    do i = 1, repetitions
        call assemble_reverse(matrix_bar, control_points_bar, weights_bar)
    end do
    call cpu_time(analytical_vjp_seconds)
    analytical_vjp_seconds = &
        (analytical_vjp_seconds - start_time)/real(repetitions, dp)
    adjoint_derivative = sum(control_points_bar*control_points_dot) + &
        sum(weights_bar*weights_dot)

    allocate (control_points_plus, mold=control_points)
    allocate (control_points_minus, mold=control_points)
    allocate (weights_plus, mold=weights)
    allocate (weights_minus, mold=weights)
    control_points_plus = control_points + difference_step*control_points_dot
    control_points_minus = control_points - difference_step*control_points_dot
    weights_plus = weights + difference_step*weights_dot
    weights_minus = weights - difference_step*weights_dot
    call cpu_time(start_time)
    do i = 1, repetitions
        call assemble_primal(control_points_plus, weights_plus, matrix_plus)
        call assemble_primal(control_points_minus, weights_minus, matrix_minus)
    end do
    call cpu_time(finite_difference_seconds)
    finite_difference_seconds = &
        (finite_difference_seconds - start_time)/real(repetitions, dp)
    allocate (solution_plus(matrix%nrow), solution_minus(matrix%nrow))
    call sparse_direct_factor_csc( &
        factor_plus, matrix_plus%nrow, matrix_plus%col_ptr, &
        matrix_plus%row_idx, matrix_plus%val, status)
    call sparse_direct_solve_factored( &
        factor_plus, rhs, solution_plus, status)
    call sparse_direct_factor_csc( &
        factor_minus, matrix_minus%nrow, matrix_minus%col_ptr, &
        matrix_minus%row_idx, matrix_minus%val, status)
    call sparse_direct_solve_factored( &
        factor_minus, rhs, solution_minus, status)
    central_derivative = dot_product( &
        solution_bar, solution_plus - solution_minus)/(2.0_dp*difference_step)
    write (*, "(a,2es16.8)") "shape derivatives: ", &
        adjoint_derivative, central_derivative
    if (abs(adjoint_derivative - central_derivative) > &
        1.0e-7_dp*max(1.0_dp, abs(central_derivative))) &
        error stop "shape adjoint failed finite-difference diagnostic"

    call make_plots()
    open (newunit=unit, file=output_directory//"/benchmark.txt", &
        status="replace", action="write")
    write (unit, "(a,i0)") "state degrees of freedom: ", matrix%nrow
    write (unit, "(a,i0)") "geometry design variables: ", &
        size(control_points) + size(weights)
    write (unit, "(a,es16.8)") "objective: ", objective
    write (unit, "(a,es16.8)") "analytical assembly JVP seconds: ", &
        analytical_jvp_seconds
    write (unit, "(a,es16.8)") "analytical assembly VJP seconds: ", &
        analytical_vjp_seconds
    write (unit, "(a,es16.8)") "two-sided assembly FD seconds: ", &
        finite_difference_seconds
    write (unit, "(a,f10.3)") "FD/JVP assembly speed ratio: ", &
        finite_difference_seconds/analytical_jvp_seconds
    write (unit, "(a,es16.8)") "adjoint directional derivative: ", &
        adjoint_derivative
    write (unit, "(a,es16.8)") "central-difference derivative: ", &
        central_derivative
    write (unit, "(a,es16.8)") "absolute derivative error: ", &
        abs(adjoint_derivative - central_derivative)
    close (unit)

    call sparse_direct_free(factor)
    call sparse_direct_free(factor_plus)
    call sparse_direct_free(factor_minus)
    call sparse_direct_free(transpose_factor)

contains

    subroutine assemble_primal(points, rational_weights, assembled)
        real(dp), intent(in) :: points(:, :, :), rational_weights(:, :)
        type(csc_t), intent(out) :: assembled

        call assemble_bspline_h1_operator_csc( &
            knots, knots, degree, degree, points, rational_weights, &
            quadrature_order, assembled, sparse_status, &
            stiffness_coefficient=1.0_dp, mass_coefficient=0.5_dp)
        if (sparse_status%code /= 0) error stop "H1 assembly failed"
    end subroutine assemble_primal

    subroutine assemble_tangent(assembled_dot)
        type(csc_t), intent(out) :: assembled_dot

        call assemble_bspline_h1_operator_csc_jvp( &
            knots, knots, degree, degree, control_points, weights, &
            control_points_dot, weights_dot, quadrature_order, assembled_dot, &
            sparse_status, stiffness_coefficient=1.0_dp, &
            mass_coefficient=0.5_dp)
        if (sparse_status%code /= 0) error stop "H1 assembly JVP failed"
    end subroutine assemble_tangent

    subroutine assemble_reverse(assembled_bar, points_bar, rational_weights_bar)
        type(csc_t), intent(in) :: assembled_bar
        real(dp), intent(out) :: points_bar(:, :, :)
        real(dp), intent(out) :: rational_weights_bar(:, :)

        call assemble_bspline_h1_operator_csc_vjp( &
            knots, knots, degree, degree, control_points, weights, &
            quadrature_order, assembled_bar, points_bar, rational_weights_bar, &
            sparse_status, stiffness_coefficient=1.0_dp, &
            mass_coefficient=0.5_dp)
        if (sparse_status%code /= 0) error stop "H1 assembly VJP failed"
    end subroutine assemble_reverse

    subroutine make_plots()
        real(dp) :: display_scale
        real(dp), allocatable :: line_x(:), line_y(:)

        allocate (displayed_points, mold=control_points)
        maximum_gradient = max(maxval(abs(control_points_bar)), tiny(1.0_dp))
        display_scale = 0.12_dp/maximum_gradient
        displayed_points = control_points - display_scale*control_points_bar
        call figure(figsize=[7.0_dp, 6.0_dp])
        allocate (line_x(size(x_points)), line_y(size(x_points)))
        do iy = 1, size(y_points)
            line_x = control_points(1, :, iy)
            line_y = control_points(2, :, iy)
            if (iy == 1) then
                call plot(line_x, line_y, label="original", &
                    linestyle="-", color=blue)
            else
                call plot(line_x, line_y, linestyle="-", color=blue)
            end if
            line_x = displayed_points(1, :, iy)
            line_y = displayed_points(2, :, iy)
            if (iy == 1) then
                call plot(line_x, line_y, label="adjoint descent step", &
                    linestyle="--", color=orange)
            else
                call plot(line_x, line_y, linestyle="--", color=orange)
            end if
        end do
        deallocate (line_x, line_y)
        allocate (line_x(size(y_points)), line_y(size(y_points)))
        do ix = 1, size(x_points)
            line_x = control_points(1, ix, :)
            line_y = control_points(2, ix, :)
            call plot(line_x, line_y, linestyle="-", color=blue)
            line_x = displayed_points(1, ix, :)
            line_y = displayed_points(2, ix, :)
            call plot(line_x, line_y, linestyle="--", color=orange)
        end do
        call xlabel("physical x")
        call ylabel("physical y")
        call title("NURBS control mesh: original and adjoint descent step")
        call legend()
        call savefig(output_directory//"/shape_gradient_2d.png")

        indices = [(real(i, dp), i=1, size(solution))]
        call figure(figsize=[8.0_dp, 5.0_dp])
        call plot(indices, solution, label="state", linestyle="-", color=blue)
        call plot(indices, solution_dot, label="directional derivative", &
            linestyle="--", color=orange)
        call xlabel("state degree of freedom")
        call ylabel("coefficient")
        call title("State and analytical geometry-to-state JVP")
        call legend()
        call savefig(output_directory//"/state_sensitivity_1d.png")

        allocate (geometry_gradient(size(control_points) + size(weights)))
        geometry_gradient(:size(control_points)) = &
            reshape(control_points_bar, [size(control_points)])
        geometry_gradient(size(control_points) + 1:) = &
            reshape(weights_bar, [size(weights)])
        indices = [(real(i, dp), i=1, size(geometry_gradient))]
        call figure(figsize=[8.0_dp, 5.0_dp])
        call plot(indices, max(abs(geometry_gradient), epsilon(1.0_dp)))
        call set_yscale("log")
        call xlabel("geometry design variable")
        call ylabel("absolute objective gradient (log scale)")
        call title("Reverse-mode gradient over all NURBS design variables")
        call savefig(output_directory//"/geometry_gradient_1d.png")
    end subroutine make_plots

    pure function greville_abscissae(knot_vector, spline_degree) result(points)
        real(dp), intent(in) :: knot_vector(:)
        integer, intent(in) :: spline_degree
        real(dp), allocatable :: points(:)
        integer :: basis

        allocate (points(size(knot_vector) - spline_degree - 1))
        do basis = 1, size(points)
            points(basis) = sum( &
                knot_vector(basis + 1:basis + spline_degree))/ &
                real(spline_degree, dp)
        end do
    end function greville_abscissae

end program iga_shape_sensitivity
```
